## 3. Visual Element Specifications (具体化：パラメータ)

### 3.1 `stella_bg.frag` (Impeller-Compatible Shader)

_Note: This shader must manage Aether warping, Element Marriage, and Y-coordinate correction for Impeller._

OpenGL Shading Language

```
#include <flutter/runtime_effect.glsl>

uniform float uTime;
uniform vec2 uSize;
uniform vec4 uElementColor;  // Base Seed Element (70%)
uniform vec4 uVibeColor;     // Daily Vibe (20%)
uniform float uProgress;     // 28-day progress (0.0-1.0)
uniform float uAetherMix;    // Aether activation (0.0 or 1.0)
uniform float uFlipY;        // Context-specific Y-axis flip (e.g., 1.0 for Impeller, 0.0 for others)

out vec4 fragColor;

mat2 rotate(float a) { return mat2(cos(a), -sin(a), sin(a), cos(a)); }

void main() {
    vec2 fragCoord = FlutterFragCoord().xy;
    // 120% Technical Precision: Impeller Coordinate Correction
    if (uFlipY > 0.0) { fragCoord.y = uSize.y - fragCoord.y; }
    vec2 uv = fragCoord / uSize.xy;
    vec2 p = (uv * 2.0 - 1.0) * vec2(uSize.x / uSize.y, 1.0);

    // Prismatic Aether Logic (Only when uAetherMix is high)
    vec3 aether = vec3(0.0);
    if(uAetherMix > 0.5) {
        for(float i = 1.0; i < 4.0; i++){
            uv = rotate(uTime * 0.1) * uv;
            aether.r += abs(0.1 / sin(uv.x + uTime * i * 0.2));
            aether.g += abs(0.1 / sin(uv.y + uTime * i * 0.3));
            aether.b += abs(0.1 / sin(uv.x + uv.y + uTime * i * 0.1));
        }
    }

    // Marriage Logic
    vec3 base = mix(uElementColor.rgb, uVibeColor.rgb, 0.3);
    float pulse = sin(uTime + uProgress * 6.28) * 0.5 + 0.5;
    vec3 evolved = base * (0.8 + pulse * uProgress);

    // Final blending
    vec3 finalColor = mix(evolved, aether * 0.2, uAetherMix);
    finalColor = mix(vec3(0.02, 0.02, 0.05), finalColor, 0.6); // Deep space base
    
    fragColor = vec4(finalColor, 1.0);
}
```
